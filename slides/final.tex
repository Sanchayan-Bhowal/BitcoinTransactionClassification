\begin{frame}{The Problem}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Goal.} Predict fraudulent Yelp reviews using both:
\begin{itemize}
    \item review-level covariates,
    \item graph dependence between related reviews.
\end{itemize}

\vspace{0.3em}
\textbf{Dataset.}
\begin{itemize}
    \item 46k reviews.
    \item 32 anonymized features.
    \item Roughly $15\%$ fraud, $85\%$ legitimate.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Three relation graphs.}
\begin{itemize}
    \item \texttt{net\_rur}: same user.
    \item \texttt{net\_rsr}: same product and star rating.
    \item \texttt{net\_rtr}: same product and month.
\end{itemize}

\vspace{0.4em}
\begin{block}{Modeling question}
Can we use the three networks directly, instead of hiding them inside graph embeddings?
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{GNN/GAT approach}
Learns network embeddings through message passing. Effective, but relation effects are harder to interpret.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Our approach}
Put the three networks directly into the probabilistic model, with one learned dependence strength per relation.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------------------
\begin{frame}{Model: weighted multi-relation Ising model}
Encode labels as spins:
\[
    \sigma_i=-1 \quad \text{fraud},\qquad \sigma_i=+1 \quad \text{legitimate}.
\]

\vspace{-0.5em}
\begin{block}{Joint model}
\[
\Pbb_{\beta,\theta}(\sigma\mid X,G_1,G_2,G_3)
\propto
\exp\left\{
\sum_{r=1}^3 \beta_r \sum_{i,j}J_{ij}^{(r)}\sigma_i\sigma_j
+2\sum_i \sigma_i b_\theta(X_i)
\right\}.
\]
\end{block}

\vspace{-0.2em}
\begin{itemize}
    \item $b_\theta(X_i)$ learns the review-level external field from covariates.
    \item $\beta_r$ measures the dependence strength of relation $r$.
    \item $J_{ij}^{(r)}$ includes feature-similarity weighting and degree normalization.
\end{itemize}

\vspace{0.2em}
\begin{alertblock}{Interpretability}
Unlike a black-box graph embedding, $\beta_{\texttt{rur}}$, $\beta_{\texttt{rsr}}$, and $\beta_{\texttt{rtr}}$ directly quantify which networks matter.
\end{alertblock}
\end{frame}

%------------------------------------------------------------
\begin{frame}{Fitting and posterior prediction}
\textbf{Training.} Estimate $(\widehat\beta,\widehat\theta)$ by pseudo-likelihood, optimized with AdamW.

\textbf{Posterior fraud marginal:}
\[
\widehat p_i
=\frac1M\sum_{m=1}^M
\ind\{\sigma_i^{(m)}=-1\}.
\]

\vspace{0.7em}
We sample full label configurations using the \textbf{Swendsen-Wang algorithm}, which is substantially faster than single-site updates.
\end{frame}

%------------------------------------------------------------
\begin{frame}{Takeaways}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Modeling.}
\begin{itemize}
    \item Directly encode three Yelp relation graphs.
    \item Use covariates to learn the external field.
    \item Use relation-specific parameters to interpret network importance.
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
\textbf{Results.}
\begin{itemize}
    \item Swendsen-Wang marginals improve weighted models.
    \item \texttt{net\_rur} is the most important network.
    \item The model is faster and more interpretable than GCN/GAT baselines while improving F1.
\end{itemize}
\end{column}
\end{columns}

\vspace{0.8em}
\begin{alertblock}{Final message}
Graph dependence is not just useful for prediction; modeling it directly also tells us which relationships matter for fraud detection.
\end{alertblock}
\end{frame}

                \mathbb{P}(\sigma_i=-1\mid \sigma_{-i})

            Hence large $|H_i|$ pushes the conditional probability toward $0$ or $1$.

            \vspace{0.4em}
            The scatter shows the expected pattern: high posterior uncertainty appears mostly when the fitted local field is small.
        \end{column}

    \vspace{0.35em}
    \small
    This supports the variance interpretation: nodes near the decision boundary are precisely those with weak fitted local evidence.
