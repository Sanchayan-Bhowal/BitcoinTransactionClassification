\begin{frame}{Pairwise Markov Random Field Motivation}
  Drop the conditional-independence assumption: let neighboring labels be \emph{dependent}.

  \vspace{0.5em}
  For a single graph $G=(V,E)$, consider:
  \[
    \mathbb{P}(\sigma_1,\ldots,\sigma_n \mid X, G)
    \propto
    \prod_{(i,j)\in E} \psi_{X_i, X_j}(\sigma_i, \sigma_j),
    \quad \sigma_i \in \{-1, +1\}
  \]

  \vspace{0.3em}
  Any symmetric log-potential factors as:
  \[
    \log \psi_{ij}(\sigma_i,\sigma_j)
    = \underbrace{\beta_{ij}\,\sigma_i\sigma_j}_{\text{edge weights}} + \underbrace{\alpha_{ij}(\sigma_i+\sigma_j)}_{\text{node weights}} + c_{ij}
  \]

  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{block}{$\beta_{ij} > 0$}
        Adjacent labels prefer to \textbf{agree} (homophily)
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{$\beta_{ij} < 0$}
        Adjacent labels prefer to \textbf{disagree} (heterophily)
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Model: weighted multi-relation Ising model}

\begin{block}{Joint model}
\[
\Pbb_{\beta,\theta}(\sigma\mid X,G_1,G_2,G_3)
\propto
\exp\left\{
\sum_{r=1}^3 \beta_r \sum_{(i,j) \in E_r}J_{ij}^{(r)}\sigma_i\sigma_j
+2\sum_i \sigma_i b_\theta(X_i)
\right\}.
\]
\end{block}

\vspace{-0.2em}
\begin{itemize}
    \item $b_\theta(X_i)$ learns the review-level external field from covariates.
    \item $\beta_r$ measures the dependence strength of relation $r$.
    \item $J_{ij}^{(r)}$ includes feature-similarity weighting and degree normalization.
\end{itemize}

\vspace{0.2em}
\begin{alertblock}{Interpretability}
Unlike a black-box graph embedding, $\beta_{\texttt{rur}}$, $\beta_{\texttt{rsr}}$, and $\beta_{\texttt{rtr}}$ directly quantify which networks matter.
\end{alertblock}
\end{frame}

%------------------------------------------------------------
\begin{frame}{Fitting and posterior prediction}
\textbf{Training.} Estimate $(\widehat\beta,\widehat\theta)$ by pseudo-likelihood, optimized with AdamW.

\textbf{Posterior fraud marginal:}
\[
\widehat p_i
=\frac1M\sum_{m=1}^M
\ind\{\sigma_i^{(m)}=-1\}.
\]

\vspace{0.7em}
We sample full label configurations using the \textbf{Swendsen-Wang algorithm}, which is substantially faster than single-site updates.
\end{frame}

\begin{frame}{Takeaways}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Modeling.}
\begin{itemize}
    \item Directly encode three Yelp relation graphs.
    \item Use covariates to learn the external field.
    \item Use relation-specific parameters to interpret network importance.
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
\textbf{Results.}
\begin{itemize}
    \item The model is faster and more interpretable than GCN/GAT baselines while improving F1.
    \item \texttt{net\_rur} is the most important network.
\end{itemize}
\end{column}
\end{columns}

\vspace{0.8em}
\begin{alertblock}{Final message}
Graph dependence is not just useful for prediction; modeling it directly also tells us which relationships matter for fraud detection.
\end{alertblock}
\end{frame}

        \begin{column}{0.38\textwidth}
            \small
            Conditional on the current neighbors,
            \[
                \mathbb{P}(\sigma_i=-1\mid \sigma_{-i})
                =\operatorname{logit}^{-1}(-2H_i).
            \]
            Hence large $|H_i|$ pushes the conditional probability toward $0$ or $1$.

            \vspace{0.4em}
            The scatter shows the expected pattern: high posterior uncertainty appears mostly when the fitted local field is small.
        \end{column}
    \end{columns}

    \vspace{0.35em}
    \small
    This supports the variance interpretation: nodes near the decision boundary are precisely those with weak fitted local evidence.
