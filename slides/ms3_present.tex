\begin{frame}{Baseline: CAR-Regularized Logistic Regression}
  \begin{block}{Model}
    \[
      \text{logit}\,\mathbb{P}(y_i = 1)
      = \ell_i + b + \gamma\, z_i^{\textsf{rur}} + u_i
    \]
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Component lifting term:}
      \[
        z_i^{\textsf{rur}} = \log\frac{\theta_i}{1-\theta_i} - \log\frac{\hat\pi}{1-\hat\pi}
      \]
      $\theta_i$ estimated via \textbf{Empirical Bayes Beta-Binomial}:
      \[
        \theta_i = \frac{a + k_C}{a + b + n_C}
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Graph-Laplacian (CAR) regularizer:}
      \[
        \pi(u) \propto \exp\left(-\frac{1}{2\tau^2} u^\top(I-W)u\right)
      \]
      \[
        \Leftrightarrow \quad \min\ \text{BCE} + \lambda \sum_{(i,j)\in E}(u_i - u_j)^2
      \]
    \end{column}
  \end{columns}

  \vspace{0.6em}
  \begin{alertblock}{Empirical Finding}
    The CAR prior \emph{only marginally} improves over plain logistic regression.
  \end{alertblock}
\end{frame}

\section{Main Model: Weighted Multi-graph Ising Model}
%============================================================

\begin{frame}{Pairwise Markov Random Field Motivation}
  Drop the conditional-independence assumption: let neighboring labels be \emph{dependent}.

  \vspace{0.5em}
  For a single graph $G=(V,E)$, consider:
  \[
    \mathbb{P}(\sigma_1,\ldots,\sigma_n \mid X, G)
    \propto
    \prod_{(i,j)\in E} \psi_{X_i, X_j}(\sigma_i, \sigma_j),
    \quad \sigma_i \in \{-1, +1\}
  \]

  \vspace{0.3em}
  Any symmetric log-potential factors as:
  \[
    \log \psi_{ij}(\sigma_i,\sigma_j)
    = \underbrace{\beta_{ij}\,\sigma_i\sigma_j}_{\text{edge weights}} + \underbrace{\alpha_{ij}(\sigma_i+\sigma_j)}_{\text{node weights}} + c_{ij}
  \]

  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{block}{$\beta_{ij} > 0$}
        Adjacent labels prefer to \textbf{agree} (homophily)
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{$\beta_{ij} < 0$}
        Adjacent labels prefer to \textbf{disagree} (heterophily)
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{RBF Similarity Weighting}
  \textbf{Motivation:} Nodes with \emph{similar features} should influence each other more.

  \vspace{0.5em}
  For each edge $(i,j) \in E_r$, define the \textbf{RBF weight}:
  \[
    w_{ij}^{(r)} = \exp\left(-\frac{\|X_i - X_j\|_2^2}{2\tau_r^2}\right), \quad \tau_r > 0 \text{ (bandwidth per graph)}
  \]

  \vspace{0.4em}
  \textbf{Degree normalization} prevents high-degree nodes from dominating, using weighted degree $D_i^{(r)} = \sum_k A_{ik}^{(r)} w_{ik}^{(r)}$:
  \[
    \text{Edge contribution:} \quad \beta_{ij}^{(r)}=\beta_r
    A_{ij}^{(r)} w_{ij}^{(r)}\left(\frac{1}{D_i^{(r)}} + \frac{1}{D_j^{(r)}}\right)\sigma_i\sigma_j
  \]
\end{frame}

\begin{frame}{Full Model Specification}
  \begin{block}{Weighted multi-graph Ising Model}
    \[
      \mathbb{P}_{\beta,\theta}(\sigma \mid X, \{G_i\}_{i=1}^3)
      \propto
      e^{\sum_{r=1}^{3} \beta_r \sum_{(i,j)\in E_r} w_{ij}^{(r)}\left(\frac{1}{D_i^{(r)}} + \frac{1}{D_j^{(r)}}\right)\sigma_i\sigma_j
        +
        2\sum_{i=1}^n \sigma_i b_\theta(X_i)}
    \]
  \end{block}

  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Two variants:}
      \begin{itemize}
        \item \texttt{hm\_weighted\_linear}: $b_\theta(X_i) = X_i^\top\theta$
        \item \texttt{hm\_weighted\_nn}: MLP external field
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Parameters to learn:}
      \begin{itemize}
        \item $\beta_1, \beta_2, \beta_3$: graph couplings
        \item $\theta$: feature parameters
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Training \& Inference}
%============================================================

\begin{frame}{Pseudo-Maximum Likelihood Estimation}
  \textbf{Challenge:} Normalizing constant sums over $2^n$ configurations --- intractable.

  \vspace{0.4em}
  \textbf{Solution:} Replace joint likelihood with \textbf{product of node-wise conditionals}.

  \vspace{0.3em}
  Each conditional is: $$\mathbb{P}(\sigma_i \mid X,G_i, \sigma_{-i}) = e^{\sigma_i h_i} / (2\cosh h_i)$$
  where,
  \[
    h_i = \sum_{r=1}^{3} \beta_r \sum_j A_{ij}^{(r)} w_{ij}^{(r)}\left(\frac{1}{D_i^{(r)}} + \frac{1}{D_j^{(r)}}\right)\sigma_j^{\text{obs}} + 2\,b_\theta(X_i)
  \]


  \begin{block}{Pseudo-Log-Likelihood Objective}
    \[
      \mathcal{L}(\beta, \theta) = \frac{1}{|\mathcal{T}|}\sum_{i\in\mathcal{T}} \bigl[\log(2\cosh h_i) - \sigma_i h_i\bigr]
    \]
  \end{block}
\end{frame}

\begin{frame}{Prediction \& Threshold Selection}
  Fitted probability that node $i$ is \emph{legitimate} ($\sigma_i = +1$):
  \[
    \hat p_i
    = \operatorname{logit}^{-1}\left[2\left(\sum_{r=1}^3 \hat\beta_r A_{ij}^{(r)} w_{ij}^{(r)}\left(\tfrac{1}{D_i^{(r)}}+\tfrac{1}{D_j^{(r)}}\right)\sigma_j^{\text{obs}} + 2b_{\hat\theta}(X_i)\right)\right]
  \]

  \textbf{Binary prediction:}
      \[
        \hat\sigma_i = \begin{cases} +1 & \hat{p}_i \ge t \\ -1 & \text{otherwise} \end{cases}
      \]
      Threshold $t$ tuned on \textbf{validation set} to maximize F1.
\end{frame}
