\section{Quick Recap}
%============================================================

\begin{frame}{Dataset Recap}
    \textbf{FraudYelpDataset} --- graph of Yelp reviews
    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Three relation types:}
            \begin{itemize}
                \item \texttt{net\_rur}: same user
                \item \texttt{net\_rsr}: same product + star rating
                \item \texttt{net\_rtr}: same product + same month
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Scale:}
            \begin{itemize}
                \item 46k nodes ($85\%$ legit, $15\%$ fraud)
                \item 32 anonymized features per node
                \item 99k \texttt{rur}, 1.1M \texttt{rsr}, 6.8M \texttt{rtr} edges
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \begin{alertblock}{Key homophily finding (from last week)}
        \texttt{net\_rur} is near-perfect assortative: $P(\text{fraud} \mid \text{fraud neighbor}) = 0.909$
    \end{alertblock}
\end{frame}

\begin{frame}{Model Recap: Weighted Multi-Relation Ising}
    Encode label as spin $\sigma_i \in \{-1,+1\}$ ($-1 = $ fraud).

    \vspace{0.3em}
    \textbf{RBF-weighted, degree-normalized interaction:}
    \[
        w_{ij}^{(r)} = \exp\!\left(-\frac{\|X_i-X_j\|_2^2}{2\tau_r^2}\right),
        \qquad
        J_{ij}^{(r)} = A_{ij}^{(r)} w_{ij}^{(r)}\!\left(\tfrac{1}{D_i^{(r)}}+\tfrac{1}{D_j^{(r)}}\right)
    \]

    \begin{block}{Joint distribution}
        \[
            \mathbb{P}_{\beta,\theta}(\sigma \mid X, G_1,G_2,G_3)
            \propto
            \exp\!\Bigl\{
              \sum_{r=1}^3 \beta_r \sum_{i,j} J_{ij}^{(r)}\sigma_i\sigma_j
              + 2\sum_i \sigma_i\, b_\theta(X_i)
            \Bigr\}
        \]
    \end{block}

    \vspace{0.2em}
    Fitted via \textbf{pseudo-maximum likelihood} (AdamW). Two variants:
    \texttt{hm\_weighted\_linear} and \texttt{hm\_weighted\_nn}.

    \vspace{0.3em}
    \begin{alertblock}{Last week's limitation}
        Prediction used the \emph{local conditional} $p_i^{\mathrm{cond}} = \mathbb{P}(\sigma_i=-1 \mid \sigma_{\mathcal{T}}, X, G)$, not the true posterior marginal.
    \end{alertblock}
\end{frame}

\section{This Week: MCMC Marginals}
%============================================================

\begin{frame}{Criticism \& Revision}
    \textbf{Problem:} $p_i^{\mathrm{cond}}$ is a \emph{local} conditional --- it fixes all observed labels and ignores uncertainty in the test nodes' labels.

    \vspace{0.5em}
    \textbf{What we really want:} the posterior \emph{marginal}
    \[
        p_i^{\mathrm{marg}}
        =
        \mathbb{P}_{\widehat\beta,\widehat\theta}(\sigma_i = -1 \mid X, G)
    \]
    which averages over the joint distribution of all unobserved labels.

    \vspace{0.5em}
    \textbf{Solution:} Estimate $p_i^{\mathrm{marg}}$ via \textbf{Gibbs sampling} on the fitted Ising model.

    \vspace{0.4em}
    The Gibbs update for node $i$:
    \[
        H_i(\sigma_{-i})
        =
        \sum_{r=1}^3 \widehat\beta_r \sum_{j \ne i} J_{ij}^{(r)}\sigma_j
        + 2\,b_{\widehat\theta}(X_i),
        \qquad
        \mathbb{P}(\sigma_i=-1 \mid \sigma_{-i}) = \operatorname{logit}^{-1}(-2H_i)
    \]
\end{frame}

\begin{frame}{MCMC Estimation Procedure}
    After estimating $(\widehat\beta, \widehat\theta)$ by pseudo-likelihood:

    \vspace{0.4em}
    \begin{enumerate}
        \item Run Gibbs sampling with the fitted Ising model to generate $M$ full configurations
        $\sigma^{(1)},\ldots,\sigma^{(M)}\in\{-1,+1\}^N$.
        \item Estimate marginal fraud probability per node:
        \[
            \widehat{p}_i = \frac{1}{M}\sum_{m=1}^{M} \mathbbm{1}\{\sigma_i^{(m)} = -1\}.
        \]
        \item Threshold $\widehat{p}_i^{\mathrm{MCMC}}$ on validation F1 for binary prediction.
    \end{enumerate}

    \vspace{0.5em}
    \begin{block}{Why this is better}
        The marginal naturally incorporates the global structure of the Ising model across all nodes, not just local conditional information at each test node.
    \end{block}
\end{frame}

\begin{frame}{MCMC Results: F1 Comparison}
    \begin{figure}
        \centering
        \includegraphics[width=\linewidth]{figures/fraud_yelp/f1_comp_mcmc.png}
        \caption{F1-score: MCMC marginal vs.\ previous conditional scoring.}
    \end{figure}
\end{frame}

\begin{frame}{MCMC Results: Key Takeaway}
    \begin{block}{Gains from MCMC revision}
        \begin{itemize}
            \item Weighted neural (\texttt{hm\_weighted\_nn}): $+0.028$ F1
            \item Weighted linear (\texttt{hm\_weighted\_linear}): $+0.025$ F1
            \item Marginal scoring consistently outperforms conditional scoring for weighted models
        \end{itemize}
    \end{block}

    \vspace{0.5em}
    The improvement confirms that the full Ising joint structure carries signal beyond the local conditional --- the revision is warranted.
\end{frame}

\begin{frame}{Uncertainty Analysis Conclusions}
    \begin{itemize}
        \item Smaller $\widehat p_i(1-\widehat p_i)$ means the posterior marginal is closer to $0$ or $1$.
        \item HM weighted variants give more concentrated marginals than GM weighted variants.
        \item Edge ablations show that \texttt{rur} is the main predictive relation, while adding \texttt{rsr} or \texttt{rtr} can still reduce uncertainty.
        \item Lower variance is a diagnostic, not a standalone objective: it must be read together with F1 and average precision.
    \end{itemize}

    \vspace{0.4em}
    \begin{block}{Takeaway}
        MCMC marginals let us evaluate both prediction quality and posterior uncertainty under the fitted Ising model.
    \end{block}
\end{frame}

\begin{frame}{Backup: Local Field Explains Posterior Uncertainty}
    \begin{columns}[T]
        \begin{column}{0.60\textwidth}
            \includegraphics[width=\linewidth]{figures/fraud_yelp/local_field_vs_uncertainty.png}
        \end{column}
        \begin{column}{0.38\textwidth}
            \small
            Conditional on the current neighbors,
            \[
                \mathbb{P}(\sigma_i=-1\mid \sigma_{-i})
                =\operatorname{logit}^{-1}(-2H_i).
            \]
            Hence large $|H_i|$ pushes the conditional probability toward $0$ or $1$.

            \vspace{0.4em}
            The scatter shows the expected pattern: high posterior uncertainty appears mostly when the fitted local field is small.
        \end{column}
    \end{columns}

    \vspace{0.35em}
    \small
    This supports the variance interpretation: nodes near the decision boundary are precisely those with weak fitted local evidence.
\end{frame}

\begin{frame}{Backup: Variance Identity for Spins}
    Let
    \[
        \widehat\mu_i=\frac1M\sum_{m=1}^M \sigma_i^{(m)},
        \qquad
        \widehat p_i=\frac1M\sum_{m=1}^M\mathbbm 1\{\sigma_i^{(m)}=-1\}.
    \]
    Since $\sigma_i\in\{-1,+1\}$,
    \[
        \widehat p_i=\frac{1-\widehat\mu_i}{2},
        \qquad
        \widehat p_i(1-\widehat p_i)=\frac{1-\widehat\mu_i^2}{4}.
    \]

    \begin{block}{Interpretation}
        Posterior variance is small exactly when the posterior spin mean is close to $+1$ or $-1$; it is largest when the posterior spin mean is near $0$.
    \end{block}
\end{frame}
